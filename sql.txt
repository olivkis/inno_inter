--Output the number of movies in each category, sorted descending
SELECT category.name, 
       COUNT(film_id) AS movie_count
FROM film_category LEFT JOIN category using(category_id) 
GROUP BY category.name
ORDER BY movie_count DESC

  
--Output the 10 actors whose movies rented the most, sorted in descending order (NEW VERSION)!
WITH actors_rental AS (
	SELECT fa.actor_id,
		   COUNT(r.rental_id) AS rental_count
	FROM film_actor fa
	LEFT JOIN inventory i using(film_id)
	LEFT JOIN rental r using(inventory_id)
	GROUP BY fa.actor_id
)

SELECT a.first_name, 
       a.last_name
FROM actors_rental
LEFT JOIN actor a using(actor_id)
ORDER BY rental_count DESC
LIMIT 10

-- Output the category of movies on which the most money was spent
SELECT c.name, 
       SUM(p.amount) AS costs
FROM category c 
LEFT JOIN film_category fc using(category_id)
LEFT JOIN inventory i using(film_id)
LEFT JOIN rental r using(inventory_id)
LEFT JOIN payment p using(rental_id)
GROUP BY c.category_id
ORDER BY costs DESC
LIMIT 1

  
--Print the names of movies that are not in the inventory. 
--Write a query without using the IN operator
SELECT f.title
FROM film f LEFT JOIN inventory i using(film_id) 
WHERE i.film_id IS NULL

  
-- Output the top 3 actors who have appeared the most in movies in the “Children” category. 
-- If several actors have the same number of movies, output all of them.
WITH ranked_actors AS 
       (
		SELECT c.name,
				a.first_name,
				a.last_name,
				DENSE_RANK() OVER(ORDER BY COUNT(*) DESC) AS actors_rank
			FROM actor a 
			LEFT JOIN film_actor using(actor_id)
			LEFT JOIN film_category using(film_id)
			LEFT JOIN category c using(category_id)  
			WHERE c.name='Children' 
			GROUP BY c.name, a.actor_id)

SELECT ra.first_name,
	   ra.last_name
FROM ranked_actors ra
WHERE actors_rank <=3

  
-- Output cities with the number of active and inactive customers (active - customer.active = 1). 
-- Sort by the number of inactive customers in descending order

SELECT ct.city, 
       SUM(CASE WHEN cus.active=1 THEN 1 ELSE 0 END) AS active_users,
	   SUM(CASE WHEN cus.active=0 THEN 1 ELSE 0 END) AS inactive_users
FROM customer cus
LEFT JOIN address a using(address_id)
LEFT JOIN city ct using(city_id)
GROUP BY ct.city
ORDER BY inactive_users DESC


-- Output the category of movies that have the highest number of 
-- total rental hours in the city (customer.address_id in this city) and that 
-- start with the letter “a”. Do the same for cities that have a “-” in them. 
-- Write everything in one query.

WITH city_rental_hour AS (
    SELECT c.name, 
	      ct.city,
		  ROUND(SUM(EXTRACT(epoch from (COALESCE(return_date, NOW())-rental_date))/3600), 2) AS rental_hours
	FROM film_category fc
	LEFT JOIN category c USING(category_id)
	LEFT JOIN inventory i using(film_id)
	LEFT JOIN rental rent USING(inventory_id)
	LEFT JOIN customer cus USING(customer_id)
	LEFT JOIN address using(address_id)
	LEFT JOIN city ct using(city_id)
	GROUP BY c.name, ct.city
	), 
	
	category_types AS (
	SELECT CASE 
		       WHEN LOWER(city) LIKE 'a%' THEN 'starts_with_a'
			   WHEN city LIKE '%-%' THEN 'has_dash'
			   END AS category_type, 
		   name,	
		   SUM(rental_hours) AS rental_hours
	FROM city_rental_hour
	WHERE LOWER(city) LIKE 'a%' OR city LIKE '%-%'
	GROUP BY category_type, name
	),

	ranked_category AS (
	SELECT category_type, 
		   name, 
		   rental_hours,
		   RANK() OVER(PARTITION BY category_type ORDER BY rental_hours) AS rnk
	FROM category_types
	)
	
SELECT name, 
       category_type
FROM ranked_category
WHERE rnk=1
